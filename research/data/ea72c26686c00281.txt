from pathlib import Path
import json,re

BASE=Path(__file__).resolve().parent
catalog=json.loads((BASE/'74条公开作品目录.json').read_text())
ranking=json.loads((BASE/'各类型前三候选与分类依据.json').read_text())
text=(BASE/'分类案例分析素材.txt').read_text()
parts=re.split(r'^\[(\d+)\]\s*$',text,flags=re.M)
fields=['选题与开头','文案推进','拍摄','剪辑','商业承接','你怎么练','需要补核']
cases={}
for i in range(1,len(parts),2):
    index=int(parts[i])
    values=dict(line.split('：',1) for line in parts[i+1].strip().splitlines() if '：' in line)
    assert set(values)==set(fields),(index,values.keys())
    work=catalog['items'][index-1].copy()
    folder=BASE/'内部分析素材'/work['video_id']
    source=json.loads((folder/'来源.json').read_text())
    transcript=json.loads((folder/'自动转写-未听审.json').read_text())
    assert transcript.get('segments')
    work.update(analysis=values,duration_seconds=source['duration'],
                evidence_status='全时长自动转写与间隔画面结构分析；没有完成原声听审',
                transcript_path=str(folder/'自动转写-未听审.json'))
    if (folder/'短片细读画面.jpg').exists():
        visuals=[str(folder/'短片细读画面.jpg')]
    elif index in [1,67]:
        visuals=[str(p) for p in sorted(folder.glob('画面索引-*.jpg'))]
    else:
        visuals=[str(p) for p in sorted(folder.glob('类型研究画面-*.jpg'))]
    assert visuals,index
    work['visual_index_paths']=visuals
    cases[index]=work
selected={x['index'] for g in ranking['categories'] for x in g['selected']}
assert selected==set(cases),(selected-set(cases),set(cases)-selected)

    n=round(value)
    return f'{n//60}:{n%60:02d}'

Vlog该怎样继续核实

先看是否以“人物实际经历的过程”为主线：时间、行动、遇到的问题、选择和变化是否连续。
若只是换多个场景介绍产品，按场景式种草分析更准确。
若以后找到独立Vlog，仍需按同一账号身份、公开点赞、时间与作品来源重新排名；本版不以别人的Vlog补齐。
已经整理出的生活场景类有三个候选，可先学习任务怎样被写成可拍摄的场景。

你的实操顺序

第一条先练教程：选自己确实用过的一项功能，用原素材做出结果；拍清入口、操作与完成画面。
第二条练对比：固定同一份素材与任务，先定义条件，再拍两种方法；保留不利结果。
第三条练场景：拍一次真实任务从开始到完成，产品只在有用的位置出现。
不用为了作业购买礼盒、手机或打印机。选择自己手边能用、能借、能验证的工具即可。

每条交付五样东西：
一页选题卡：给谁、解决什么、结果是什么、证据是什么、为什么此时发布。
一份带画面栏脚本：口播意图、动作、界面、素材来源、需要证明的事。
一份镜头与道具表：每句关键判断都有相应镜头或资料。
一条成片：让没看过脚本的人也能看懂过程与结果。
一页复盘：哪里看不懂，哪句超出了证据，改了什么，下一次怎样做。

审稿问题

开头的结果是否真实？步骤能否照做？画面是否证明了口播？
把某一台设备的表现推广到整类产品了吗？
把预测、厂家资料、网络素材和个人实测混在一起了吗？
优惠是否写明时间和条件？哪些主张还需要来源？
有没有因为某条高赞，就把个人猜测写成了它变火的确定原因？
