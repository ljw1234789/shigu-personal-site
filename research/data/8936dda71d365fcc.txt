from pathlib import Path

BASE = Path(__file__).resolve().parent
PACK = BASE / '求职准备总包'
MEDIA = BASE / '内部分析素材'

def read(name):
    return json.loads((PACK / name).read_text())

def save(name, obj):
    (PACK / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2))

assert len(cases) == len({case['video_id'] for case in cases}) == 32

by_id = {case['video_id']: case for case in cases}
ledger = read('09_74条研究覆盖与缺口.json')
metadata = {}
for row in ledger['items']:
    vid = row['video_id']
    folder = MEDIA / vid
    assert list(folder.glob('*.mp4')), vid
    assert (folder / '自动转写-未听审.json').is_file(), vid
    boards = sorted(folder.glob('类型研究画面-*.jpg'))
    boards += sorted(folder.glob('画面索引-*.jpg'))
    boards += sorted(folder.glob('短片2秒画面索引.jpg'))
    assert boards, vid
    meta_path = folder / '公开接口作品信息-2026-11-02.json'
    meta = json.loads(meta_path.read_text())
    assert meta['id'] == vid
    assert str(meta['author'].get('unique_id') or meta['author'].get('short_id')) == '815387110'
    metadata[vid] = meta
    row.update(structured_research_complete=True,
               review_status='结构研究：全时长自动文稿已读、间隔画面已检查；未完整连续视听或逐句听审',
               automatic_transcript_path=str(folder / '自动转写-未听审.json'),
               sampled_visual_paths=[str(path) for path in boards],
               public_metadata_path=str(meta_path),
               publish_time_Asia_Shanghai=meta['publish_time_Asia_Shanghai'],
               duration_seconds=meta['duration'],
               likes_exact_snapshot_20261102=meta['statistics_snapshot'].get('digg_count'),
               fully_audiovisually_reviewed=False, transcript_listened_and_approved=False)
    if vid in by_id:
        row['analysis_path'] = str(PACK / '10_剩余32条结构研究.txt')
        row['analysis_index'] = row['index']
        row['topic_summary_original_catalog'] = row['topic_summary']
        row['topic_summary'] = by_id[vid]['title']
