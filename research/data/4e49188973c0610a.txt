"""验证本地Omni是否真的使用声音和画面；未通过不扩大为全量验收。"""
from pathlib import Path
import json,os,sys,time,wave,subprocess
BASE=Path(__file__).resolve().parent
OUT=BASE/'原声核验'
sys.path[:0]=[str(OUT/'Omni依赖'),str(OUT/'依赖'),
    '/Users/ljw/Documents/ChatGPT/小工具/上戏搭子/runtime/voice-python',str(OUT/'RapidOCR依赖')]
os.environ.update(HF_HUB_OFFLINE='1',HF_HUB_DISABLE_TELEMETRY='1',HF_HUB_DISABLE_IMPLICIT_TOKEN='1',TOKENIZERS_PARALLELISM='false')
MODEL='/Users/ljw/.cache/huggingface/hub/models--LiquidAI--LFM3-Omni-2B-MLX-8bit/snapshots/f3b873aaa6b253e60eadb5f2388db53b84b9a5aa'
vid='7610755319857286463';folder=BASE/'内部分析素材'/vid
original_audio=folder/'audio.wav'
with wave.open(str(original_audio)) as w:
    params=w.getparams();n=w.getnframes();first=w.readframes(64000)
for path,frames in [(OUT/'omni_C.wav',first),(OUT/'omni_silence_full.wav',b'\x00'*(n*2))]:
    with wave.open(str(path),'wb') as w:w.setparams(params);w.writeframes(frames)
proxy=OUT/'联合视听样本.mp4'
if not proxy.exists():
    src=next(folder.glob('*.mp4'))
    subprocess.run(['/Users/ljw/Documents/ChatGPT/小工具/batch-media-tool/bin/ffmpeg','-nostdin','-v','error','-y','-i',str(src),'-an','-vf','scale=224:398','-c:v','libx264','-crf','22','-preset','veryfast',str(proxy)],check=True)
import mlx.core as mx
from mlx_vlm import load,generate
from mlx_vlm.prompt_utils import apply_chat_template
from mlx_vlm.generate.video import processor_handles_video
mx.set_memory_limit(7*1024**3);mx.set_cache_limit(128*1024**2)
started=time.time();print(json.dumps({'stage':'loading_omni'}),flush=True)
model,processor=load(MODEL,trust_remote_code=False)
cap={'stage':'loaded','model_type':model.config.model_type,'native_video_processor':processor_handles_video(processor),'processor_class':type(processor).__name__,'seconds':round(time.time()-started,2)}
print(json.dumps(cap),flush=True)
tasks=[
 ('A_silence',OUT/'control_A.wav',None,'Describe only the audible contents. Is there speech, music or another sound? If silent, say silent. Do not invent sounds.'),
 ('B_tone',OUT/'control_B.wav',None,'Describe only the audible contents. Is there speech, music or another sound? If silent, say silent. Do not invent sounds.'),
 ('C_speech',OUT/'omni_C.wav',None,'Transcribe the audible speech in its original language. Output only words you hear. If no intelligible speech, say so.'),
 ('D_original_av',original_audio,proxy,'Describe the sequence of visible scenes and the audible contents separately. State when speech and music occur. Base sounds only on audio, not on subtitles or likely visual events. If uncertain, say uncertain. Do not guess product specifications or song names.'),
 ('E_muted_av',OUT/'omni_silence_full.wav',proxy,'Describe the sequence of visible scenes and the audible contents separately. State when speech and music occur. Base sounds only on audio, not on subtitles or likely visual events. If uncertain, say uncertain. Do not guess product specifications or song names.')]
rows=[]
for key,audio,video,prompt in tasks:
    start=time.time();video_args={'video':[str(video)],'fps':2.0} if video else {}
    formatted=apply_chat_template(processor,model.config,prompt,num_audios=1,**video_args)
    result=generate(model,processor,formatted,audio=[str(audio)],video=[str(video)] if video else None,
        fps=2.0,max_tokens=220,temperature=0,verbose=False,prefill_step_size=256)
    row={'sample':key,'audio':str(audio),'video':str(video) if video else None,
         'video_sampling_fps':2.0 if video else None,'machine_observation':result.text,
         'prompt_tokens':getattr(result,'prompt_tokens',None),'inference_seconds':round(time.time()-start,2),
         'human_listened_and_approved':False}
    rows.append(row);(OUT/'联合视听模型盲测_原始结果.json').write_text(json.dumps({'capability':cap,'model':MODEL,'results':rows},ensure_ascii=False,indent=2))
    print(json.dumps(row,ensure_ascii=False),flush=True)
print(json.dumps({'stage':'finished','seconds':round(time.time()-started,2)}),flush=True)
