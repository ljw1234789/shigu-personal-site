TASK_PYTHON="/Users/ljw/.cache/codex-runtimes/codex-primary-runtime/dependencies/python/bin/python3"

  TASK_PYTHON="/usr/bin/python3"
fi

  read -r
fi
exit "$TASK_RESULT"
