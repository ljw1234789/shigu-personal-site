"""仅本机研究；自动转写与5秒间隔画面，不代表完整视听审阅。"""
from pathlib import Path
import subprocess, sys, json, math

BASE = Path(__file__).resolve().parent
ENGINE = Path('/Users/ljw/Documents/ChatGPT/小工具/batch-media-tool')
FFMPEG = ENGINE / 'bin/ffmpeg'
ASR = Path.home() / 'Library/Application Support/拾片/.data/asr'

def transcribe(vid):
    folder = BASE / '内部分析素材' / vid
    source = next(folder.glob('*.mp4'))
    output = folder / '自动转写-未听审.json'
    if output.exists():
        return
    audio = folder / 'audio.wav'
    subprocess.run([str(FFMPEG), '-nostdin', '-v', 'error', '-y', '-i', str(source),
                    '-vn', '-ar', '16000', '-ac', '1', str(audio)], check=True)
    subprocess.run(['/usr/bin/python3', str(ENGINE / 'engine/feishu_transcribe.py'),
                    str(ASR / 'site-packages'), str(ASR / 'ggml-small.bin'), str(audio), str(output)],
                   stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=True)
    print(json.dumps({'id':vid, 'asr':'complete_unreviewed'},ensure_ascii=False),flush=True)

def boards(vid):
    from PIL import Image, ImageDraw, ImageFont
    folder = BASE / '内部分析素材' / vid
    source = next(folder.glob('*.mp4'))
    frames = folder / '5秒间隔画面'
    frames.mkdir(exist_ok=True)
    subprocess.run([str(FFMPEG), '-nostdin', '-v', 'error', '-y', '-i', str(source),
                    '-vf', 'fps=1/5:start_time=0,scale=270:480', '-q:v', '3',
                    str(frames / '%03d.jpg')], check=True)
    files = sorted(frames.glob('*.jpg'))
    font = ImageFont.truetype('/System/Library/Fonts/Monaco.ttf',18)
    pages = []
    for start in range(0,len(files),20):
        batch = files[start:start+20]
        sheet = Image.new('RGB',(4*282,math.ceil(len(batch)/4)*516),(30,30,30))
        draw = ImageDraw.Draw(sheet)
        for i,path in enumerate(batch):
            x = (i%4)*282+6
            y = (i//4)*516+30
            t = (start+i)*5
            draw.text((x,y-25),f'~{t//60:02d}:{t%60:02d}',font=font,fill='white')
            sheet.paste(Image.open(path),(x,y))
        target = folder / f'画面索引-{start//20+1}.jpg'
        sheet.save(target,quality=90)
        pages.append(str(target))
    print(json.dumps({'id':vid,'sampledFrames':len(files),'boards':pages},ensure_ascii=False),flush=True)

if __name__ == '__main__':
    mode, *ids = sys.argv[1:]
    if mode == 'asr':
        from concurrent.futures import ThreadPoolExecutor
        with ThreadPoolExecutor(max_workers=2) as pool:
            list(pool.map(transcribe,ids))
    elif mode == 'boards':
        for vid in ids:
            boards(vid)
